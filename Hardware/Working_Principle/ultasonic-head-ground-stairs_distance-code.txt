// Head Sensor
#define TRIG_HEAD 18
#define ECHO_HEAD 19

// Front Sensor
#define TRIG_FRONT 32
#define ECHO_FRONT 33

// Stair Sensor
#define TRIG_STAIR 25
#define ECHO_STAIR 26

float getDistance(int trigPin, int echoPin) {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH, 30000);

  if (duration == 0) return 999;

  return duration * 0.0343 / 2;
}

void setup() {
  Serial.begin(115200);

  pinMode(TRIG_HEAD, OUTPUT);
  pinMode(ECHO_HEAD, INPUT);

  pinMode(TRIG_FRONT, OUTPUT);
  pinMode(ECHO_FRONT, INPUT);

  pinMode(TRIG_STAIR, OUTPUT);
  pinMode(ECHO_STAIR, INPUT);

  Serial.println("BlindNav Ultrasonic Test");
}

void loop() {

  float headDist = getDistance(TRIG_HEAD, ECHO_HEAD);
  delay(50);

  float frontDist = getDistance(TRIG_FRONT, ECHO_FRONT);
  delay(50);

  float stairDist = getDistance(TRIG_STAIR, ECHO_STAIR);

  Serial.print("HEAD: ");
  Serial.print(headDist);
  Serial.print(" cm | FRONT: ");

  Serial.print(frontDist);
  Serial.print(" cm | STAIR: ");

  Serial.print(stairDist);
  Serial.println(" cm");

  delay(300);
}
